//Получаем письмо
Письмо = Параметры.Объект;
ТипЗнч(Письмо) = ДокументСсылка.ВходящееПисьмо;

Если Письмо = Неопределено Тогда
    Возврат;
КонецЕсли;

//Входящее письмо
Если Письмо.ВидПисьма <> Перечисления.ВидыЭлектронныхПисем.Входящее Тогда
    Возврат;
КонецЕсли;

//Проверка на дубли
Если ЗначениеЗаполнено(Письмо.ВходящийДокумент) Тогда
    Возврат;
КонецЕсли;

//Создание входящего документа
ВходящийДок = Справочник.ВходящиеДокументы.СоздатьЭлемент();

//Вид Документа
ВходящийДок.ВидДокумента = Справочник.ВидыВходящихДокументов.НайтиПоНаименованию("Письмо");

//Поля для автозаполнения
ВходящийДок.Наименование = Письмо.Тема;
ВходящийДок.Содержание = Письмо.Текст;
ВходящийДок.Дата = Письмо.Дата;

//Отправитель Входящего
ВходящийДок.Отправитель = Неопределенно;
Сообщить("Контрагент не найден.")

//Кому(получатель входящего)
ВходящийДок = Справочник.СпособыДоставки.НайтиПоНаименованию("Email");

//Отвественный
Ответственный = Справочник.АдреснаяКнига.НайтиПоНаименованию("Делопроизводитель канцелярии");

Если ЗначениеЗаполнено(Ответственный) Тогда
     ВходящийДок.Ответственный = Ответственный;
Иначе
     Сообщить("Ответственный не определен автоматически");
КонецЕсли;

//Прочие реквизиты
ГрифД = Справочник.ГрифыДоступа.НайтиПоНаименованию("Общий");

Если Если ЗначениеЗаполнено(Гриф) Тогда
     ВходящийДок.Гриф = ГрифД;
Иначе 
     Сообщить("Гриф не установлен");
КонецЕсли;

ВопросДеятельности = Справочник.ВопросыДеятельности.НайтиПоНаименованию("Основная деятельность");
ВходящийДок.Вопрос = ВопросДеятельности

//Запись
ПараметрыЗаписи = новый Структура;
ПараметрыЗаписи.Вставить("Регистрация", Истина);

Попытка
    ВходящийДок.Записать(ПараметрыЗаписи);
Исключение
    Сообщить("Ошибка регистрации " + ОписаниеОшибки());
КонецПопытки;
